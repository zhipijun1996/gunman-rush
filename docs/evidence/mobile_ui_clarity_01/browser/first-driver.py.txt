def surfaces(path,color=(195,163,102)):
 a=np.array(Image.open(path).convert('RGB')).astype(int); labels,_=ndi.label(np.max(np.abs(a-color),axis=2)<3); out=[]
 for b in ndi.find_objects(labels):
  if b is None:continue
  y,x=b
  if x.stop-x.start>60 and y.stop-y.start>15:out.append([x.start,y.start,x.stop-x.start,y.stop-y.start])
 return sorted(out,key=lambda b:b[2]*b[3],reverse=True)
def game_rect(path):
 a=np.array(Image.open(path).convert('RGB'));m=a.max(2)>25
 ys=np.where(m.mean(1)>.25)[0];xs=np.where(m.mean(0)>.25)[0]
 return [int(xs[0]),int(ys[0]),int(xs[-1]+1-xs[0]),int(ys[-1]+1-ys[0])]
