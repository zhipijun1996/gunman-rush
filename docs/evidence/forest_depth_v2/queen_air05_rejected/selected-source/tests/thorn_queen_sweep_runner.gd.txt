func air_future(seconds: float) -> Vector2:
	var at := queen.motor.position
	var section: int=queen.air_section
	var leg: int=queen.air_leg
	var remaining := seconds
	if queen.motion_mode=="migration_preview": remaining=maxf(0,remaining-queen.air_clock)
	for iteration: int in 40:
		var points: Array=geometry.air_paths[section]
		var a: Array=points[leg+1];var goal := Vector2(a[0],a[1])
		var distance := at.distance_to(goal)
		var duration := distance/float(geometry.air_speed)
		if remaining<=duration: return at.move_toward(goal,remaining*float(geometry.air_speed))
		remaining-=duration;at=goal
		if leg<points.size()-2: leg+=1
		else:
			section=(section+1)%geometry.air_paths.size();leg=0
			if remaining<=float(geometry.air_preview): return at
			remaining-=float(geometry.air_preview)
	return at
