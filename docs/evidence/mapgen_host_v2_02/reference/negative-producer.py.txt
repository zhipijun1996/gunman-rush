#!/usr/bin/env python3
"""External development checks; mutations apply only to disposable copies."""
import hashlib
import json
import os
from pathlib import Path
import shutil
import subprocess
import sys

ROOT = Path(__file__).resolve().parents[1]
TOOL = Path(__file__).with_name('verify_mapgen_reference.py')
HOST = Path('/workspace/gunman-rush-refactor/resources/levels/mapgen_host_v2_01')
MEMBERS = ['map.json','content/M16.json','content/M18.json','content/M21.json']
OUT = ROOT / 'negative-01'


def write(path, data):
    path.write_text(json.dumps(data, ensure_ascii=False, indent=2) + '\n')


def resign(root):
    # Reused upstream validate_release.py::resign on test copies only. This
    # bypasses inventory errors so the shipped deeper semantic oracle is tested.
    manifest = json.loads((root / 'manifest.json').read_text())
    files = {p.relative_to(root).as_posix(): hashlib.sha256(p.read_bytes()).hexdigest()
             for p in root.rglob('*') if p.is_file() and p.name != 'manifest.json' and '__pycache__' not in p.parts}
    manifest['files'] = files
    for name in list(manifest['source']['files']):
        if name not in files:
            del manifest['source']['files'][name]
        else:
            manifest['source']['files'][name]['sha256'] = files[name]
    write(root / 'manifest.json', manifest)


def run(name, argv, expected_message):
    process = subprocess.run(argv, cwd=OUT, capture_output=True, text=True, timeout=120,
                             env={k:v for k,v in os.environ.items() if k not in ('PYTHONPATH','PYTHONHOME')})
    (OUT / (name + '.stdout.log')).write_text(process.stdout)
    (OUT / (name + '.stderr.log')).write_text(process.stderr)
    assert process.returncode == 2, (name,process.returncode,process.stdout,process.stderr)
    assert expected_message in process.stderr, (name,process.stderr)
    return {'check':name,'argv':list(map(str,argv)),'exit_code':process.returncode,
            'expected_rejection_observed':True,'expected_message':expected_message}


def main():
    OUT.mkdir(exist_ok=False)
    before = {name:(HOST/name).read_bytes() for name in MEMBERS}
    archive = ROOT / 'mapgen-python-reference-0.2.0.zip'
    changed_archive = OUT / 'changed.zip'
    changed_archive.write_bytes(archive.read_bytes()+b'changed')
    checks = [run('outer_zip_tamper',[sys.executable,'-B',str(TOOL),'--archive',str(changed_archive),
        '--host-data',str(HOST),'--out',str(OUT/'must-not-extract')],'outer ZIP hash mismatch')]
    assert not (OUT/'must-not-extract').exists()
    # A real changed motion field, rather than a cosmetic formatting mutation.
    changed_host = OUT/'changed-host'
    for name,data in before.items():
        target=changed_host/name;target.parent.mkdir(parents=True,exist_ok=True);target.write_bytes(data)
    content=json.loads((changed_host/'content/M16.json').read_text())
    assert content['moving_platforms'][0]['ellipse_x_sign']==-1
    content['moving_platforms'][0]['ellipse_x_sign']=1
    write(changed_host/'content/M16.json',content)
    checks.append(run('host_ellipse_tamper',[sys.executable,'-B',str(TOOL),'--archive',str(archive),
        '--host-data',str(changed_host),'--out',str(OUT/'host-tamper-receipt')],'reference/host file bytes differ'))
    receipt=json.loads((OUT/'host-tamper-receipt/receipt.json').read_text())
    assert sum(not row['byte_equal'] for row in receipt['comparisons'])==1
    package=ROOT/'verification-01/extracted/mapgen-python-reference-0.2.0'
    changed_package=OUT/'changed-package';shutil.copytree(package,changed_package)
    file=changed_package/'samples/success/content/M16.json'
    content=json.loads(file.read_text());content['moving_platforms'][0]['ellipse_x_sign']=1;write(file,content)
    resign(changed_package)
    checks.append(run('semantic_content_ref_tamper',[sys.executable,'-B','-I','-S',str(changed_package/'reference.py'),
        'verify','--module-zip',str(changed_package/'dependencies/map-edit-modules-1.1.0.zip')],
        'unresolved or changed content_ref:content/M16.json'))
    assert all((HOST/name).read_bytes()==data for name,data in before.items())
    result={'development_negative_checks_passed':True,'checks':checks,'host_unchanged':True,
            'mutations_only_in_external_disposable_copies':True,'godot_run':False}
    write(OUT/'receipt.json',result)
    print(json.dumps({'development_negative_checks_passed':True,'checks':len(checks),'receipt':str(OUT/'receipt.json')}))


if __name__=='__main__':main()
