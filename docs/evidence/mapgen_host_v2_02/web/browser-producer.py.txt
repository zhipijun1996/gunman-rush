"""Two real preview exits in fresh contexts; observation only, one 300s budget."""
import argparse, ast, asyncio, csv, gzip, hashlib, io, json, math, os
import re, shutil, subprocess, sys, time, urllib.request
from pathlib import Path
from urllib.parse import urljoin, urlparse
sys.dont_write_bytecode = True
SCENE = 'res://experiments/generation/entries/runtime_map_v2_slice.tscn'
PREFIX = 'MAPGEN_HOST_V2 '
RAW_ROOT = 'resources/levels/mapgen_host_v2_01/'
RAW_NAMES = ['input-lock.json','map.json','bindings.json','selection.json','content/M16.json','content/M18.json','content/M21.json']

def sha(path):
    with Path(path).open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()

def parse_args():
    p = argparse.ArgumentParser(description=__doc__)
    p.add_argument('--source-root', type=Path, required=True)
    p.add_argument('--source', required=True)
    p.add_argument('--bundle', type=Path, required=True, help='Actual bundle containing the exact PCK for byte readback')
    p.add_argument('--url', required=True, help='Root-produced preview URL; driver never injects launch arguments')
    p.add_argument('--pack-sha256', required=True)
    p.add_argument('--pack-bytes', type=int, required=True)
    p.add_argument('--expected-host-load-identity')
    p.add_argument('--producer-file', type=Path, action='append', default=[])
    p.add_argument('--output', type=Path, required=True)
    p.add_argument('--commands', type=Path, required=True)
    p.add_argument('--ocr-prefix', type=Path, required=True)
    p.add_argument('--proxy-from-environment', action='store_true')
    a = p.parse_args()
    assert re.fullmatch('[0-9a-f]{40}', a.source)
    assert re.fullmatch('[0-9a-f]{64}', a.pack_sha256) and a.pack_bytes > 0
    if a.expected_host_load_identity: assert re.fullmatch('[0-9a-f]{64}', a.expected_host_load_identity)
    assert urlparse(a.url).scheme in ('http','https')
    if urlparse(a.url).hostname not in ('127.0.0.1','localhost','::1'): assert a.proxy_from_environment
    return a

def helpers(root, ocr_prefix):
    sys.path.insert(0, str(root/'tools'))
    from verify_forest_browser import locate, locate_original, norm, validate_identity, environment_proxy
    from studio_gui_pointer import locate_pixel_control, native_surfaces, title_surface, touch_menu_surface
    from verify_ci_artifact_browser import SourceRoot
    from check_web_bundle import Pack
    from style_exploration_delivery import resource
    os.environ['TESSDATA_PREFIX'] = str(ocr_prefix)
    ns = dict(csv=csv, io=io, os=os, subprocess=subprocess, locate=locate,
              locate_original=locate_original, norm=norm, locate_pixel_control=locate_pixel_control,
              native_surfaces=native_surfaces, title_surface=title_surface,
              touch_menu_surface=touch_menu_surface,
              TESS_ARGS=['--tessdata-dir','/workspace/toolchains/tessdata-4.1.0-fast'])
    tree = ast.parse((root/'tools/verify_four_line_browser.py').read_text())
    nodes = [n for n in tree.body if isinstance(n, ast.FunctionDef) and n.name in {'locate_cjk','locate_primary','locate_any'}]
    assert len(nodes) == 3
    exec(compile(ast.Module(body=nodes,type_ignores=[]),'existing_GUI_helpers','exec'),ns)
    return ns['locate_any'], validate_identity, environment_proxy, SourceRoot, Pack, resource

async def run(a, report):
    started = time.monotonic()
    out = a.output; out.mkdir(parents=True, exist_ok=False)
    a.commands.mkdir(parents=True, exist_ok=True)
    root = a.source_root.resolve(); bundle = a.bundle.resolve()
    locate_any, validate_identity, environment_proxy, SourceRoot, Pack, resource = helpers(root, a.ocr_prefix)
    from playwright.async_api import async_playwright
    def save(): (out/'report.json').write_text(json.dumps(report,ensure_ascii=False,indent=2)+'\n')
    def compressed(name, value):
        raw = json.dumps(value,separators=(',',':')).encode()
        (out/name).write_bytes(gzip.compress(raw,mtime=0))
        return hashlib.sha256(raw).hexdigest()
    def remaining(maximum):
        left = 300-(time.monotonic()-started)
        if left <= 0: raise TimeoutError('Original total300s budget exhausted')
        return min(maximum, int(left*1000))
    report.update(status='RUNNING', source=a.source, url=a.url, scope='Two planned fresh contexts, real keyboard/mouse and passive source console observations only. No init scripts, FPS04, GL/RAF observers, game resource/state writes, teleport or performance/device acceptance. Total300s including both contexts and source checks.', sessions=[], browser_closed=False)
    save()
    guard = SourceRoot(root, a.source)
    before = guard.snapshot()
    report['source_before_raw_sha256'] = compressed('source-before.json.gz', before)
    report['source_before_file_count'] = len(before['source_sha256'])
    info = json.loads((bundle/'build-info.json').read_text())
    pack_path = bundle/info['pack']
    assert info['source_commit'] == a.source and info['build_id'] == a.pack_sha256[:12]
    assert info['pack_bytes'] == a.pack_bytes == pack_path.stat().st_size
    assert sha(pack_path) == a.pack_sha256
    pack = Pack(pack_path)
    assert resource(pack, SCENE.removeprefix('res://'))
    assert resource(pack, 'experiments/platform/entries/cloud_preview.tscn')
    report['raw_JSON_readback'] = []
    for name in RAW_NAMES:
        path = RAW_ROOT+name; raw = pack.read(path)
        assert raw == (root/path).read_bytes()
        report['raw_JSON_readback'].append(dict(path=path,bytes=len(raw),sha256=hashlib.sha256(raw).hexdigest()))
    lock = json.loads(pack.read(RAW_ROOT+'input-lock.json'))
    assert set(lock['files']) == set(RAW_NAMES)-{'input-lock.json'}
    for name,digest in lock['files'].items(): assert hashlib.sha256(pack.read(RAW_ROOT+name)).hexdigest() == digest
    frozen = [Path(__file__).resolve(),root/'tools/verify_four_line_browser.py',root/'tools/verify_forest_browser.py',root/'tools/studio_gui_pointer.py',root/'tools/verify_ci_artifact_browser.py',root/'tools/check_web_bundle.py',root/'experiments/platform/cloud_preview.gd',root/'experiments/generation/runtime_map_slice.gd',root/'experiments/generation/entries/runtime_map_v2_slice.tscn',pack_path]
    frozen += [bundle/name for name in ['index.html','build-info.json','index.cache-sw.js','build_report.json','preview.json'] if (bundle/name).is_file()]
    frozen += [p.resolve() for p in a.producer_file]
    report['producer_helpers_source_before'] = {str(p):sha(p) for p in dict.fromkeys(frozen)}
    opener = urllib.request.build_opener() if a.proxy_from_environment else urllib.request.build_opener(urllib.request.ProxyHandler({}))
    base = urljoin(a.url, './')
    def served_identity():
        nonce = str(time.time_ns())
        def read(name):
            return opener.open(urllib.request.Request(base+name+'?gui='+nonce,headers={'Cache-Control':'no-cache'}),timeout=30).read()
        metadata = json.loads(read('build-info.json')); worker = read('index.cache-sw.js').decode(); html = read('index.html').decode()
        identity = validate_identity(metadata,worker,{'sha256':a.pack_sha256,'bytes':a.pack_bytes},a.source,a.pack_sha256[:12])
        match = re.search(r'/\* cloud-preview-entry \*/ args: (\[[^\n]*\]),',html)
        assert match, 'Expected original root-produced cloud-preview entry, no driver injection fallback'
        flags = json.loads(match.group(1))
        assert '--cloud-preview' in flags and '--entry=scene' in flags and '--scene='+SCENE in flags
        return dict(identity=identity,html_sha256=hashlib.sha256(html.encode()).hexdigest(),launch_flags=flags)
    report['served_before'] = served_identity(); save(); remaining(300000)
    seq = 0
    async def checkpoint(label, pic, session):
        nonlocal seq
        seq += 1
        report['waiting'] = dict(seq=seq,label=label,screenshot=pic.name,session=session['label'])
        save(); print('WAIT',seq,label,pic.name,flush=True)
        while True:
            remaining(300000)
            command = a.commands/'command.json'
            if command.exists():
                value = json.loads(command.read_text())
                if value.get('seq') == seq: return value
            await asyncio.sleep(.1)
    async with async_playwright() as p:
        browser = await p.chromium.launch(executable_path='/usr/bin/chromium', proxy=environment_proxy()[0] if a.proxy_from_environment else None,headless=True,args=['--no-sandbox','--use-angle=swiftshader','--enable-unsafe-swiftshader'],timeout=remaining(30000))
        report['browser_version'] = browser.version
        try:
            for label,key,exit_kind in [('menu-click','d','menu-click'),('escape','a','escape')]:
                row = dict(label=label,console=[],page_errors=[],request_failures=[],http_errors=[],observations=[],parse_errors=[],screenshots=[],inputs=[])
                report['sessions'].append(row); save()
                context = await browser.new_context(viewport={'width':1280,'height':720},device_scale_factor=1,has_touch=True)
                page = await context.new_page()
                def console(message, row=row):
                    text = message.text; stamp = time.time()*1000
                    row['console'].append(dict(type=message.type,text=text,location=message.location,epoch_ms=stamp))
                    if text.startswith(PREFIX):
                        try:
                            value = json.loads(text[len(PREFIX):]); row['observations'].append(dict(epoch_ms=stamp,value=value))
                        except Exception as error: row['parse_errors'].append(repr(error))
                page.on('console',console)
                page.on('pageerror',lambda error,row=row:row['page_errors'].append(str(error)))
                page.on('requestfailed',lambda request,row=row:row['request_failures'].append(dict(url=request.url,failure=request.failure)))
                page.on('response',lambda response,row=row:row['http_errors'].append(dict(url=response.url,status=response.status)) if response.status>=400 else None)
                async def wait_for(predicate, maximum=15000):
                    deadline = time.monotonic()+remaining(maximum)/1000
                    while not predicate():
                        if time.monotonic()>=deadline: raise TimeoutError(label+' expected source observation absent')
                        await page.wait_for_timeout(100)
                async def screen(name):
                    path = out/(label+'-'+name+'.png'); await page.screenshot(path=str(path),timeout=remaining(15000))
                    row['screenshots'].append(dict(file=path.name,sha256=sha(path),epoch_ms=time.time()*1000)); save(); return path
                def physics(): return [x['value'] for x in row['observations'] if x['value'].get('event')=='physics']
                try:
                    await page.goto(a.url,wait_until='networkidle',timeout=remaining(60000))
                    await page.wait_for_function('window.PLAYTEST_LOADING_DONE===true',timeout=remaining(60000))
                    row['no_FPS04'] = await page.evaluate("typeof window.FPS04_READY==='undefined' && typeof window.FPS04_STATE==='undefined' && typeof window.FPS04_RESULTS==='undefined'")
                    assert row['no_FPS04']
                    await wait_for(lambda:any(x['value'].get('event')=='loaded' for x in row['observations']))
                    await wait_for(lambda:len(physics())>=2)
                    loaded = next(x['value'] for x in row['observations'] if x['value'].get('event')=='loaded')
                    assert re.fullmatch('[0-9a-f]{64}',loaded['host_load_identity'])
                    if a.expected_host_load_identity: assert loaded['host_load_identity']==a.expected_host_load_identity
                    row['host_load_identity'] = loaded['host_load_identity']
                    for v in [loaded,*physics()]:
                        assert v['host_load_identity']==loaded['host_load_identity'] and isinstance(v['grounded'],bool)
                        assert all(isinstance(v[field],list) and len(v[field])==2 and all(isinstance(x,(int,float)) and math.isfinite(x) for x in v[field]) for field in ['player','carrier'])
                    pic = await screen('01-loaded')
                    cmd = await checkpoint('view-loaded-before-real-input',pic,row); assert cmd['action']=='move'
                    baseline = physics()[-1]; row['input_baseline'] = baseline
                    begin = time.time()*1000; await page.keyboard.down(key); await page.wait_for_timeout(180); await page.keyboard.up(key)
                    row['inputs'].append(dict(action=key+'180',begin_epoch_ms=begin,end_epoch_ms=time.time()*1000))
                    count = len(physics()); await wait_for(lambda:len(physics())>=count+2)
                    post = physics()[-1]; row['input_after'] = post
                    row['directional_x_delta'] = post['player'][0]-baseline['player'][0]
                    assert row['directional_x_delta']*(1 if key=='d' else -1)>1.0
                    points = [x['carrier'] for x in physics()]
                    row['carrier_max_displacement'] = max(math.dist(points[0],v) for v in points)
                    assert row['carrier_max_displacement']>.01
                    row['grounded_observed'] = any(x['grounded'] for x in physics()); assert row['grounded_observed']
                    pic = await screen('02-after-input')
                    cmd = await checkpoint('view-before-planned-exit',pic,row); assert cmd['action']==exit_kind
                    if exit_kind=='menu-click':
                        point = locate_any(pic,'FULL GAME / MENU'); row['exit_point'] = point
                        await page.mouse.click(*point); row['inputs'].append(dict(action='actual-FULL-GAME-MENU-click',point=point))
                    else:
                        await page.keyboard.press('Escape'); row['inputs'].append(dict(action='actual-Escape-key'))
                    await wait_for(lambda:any(x['value'].get('event')=='closed' for x in row['observations']),30000)
                    closed = [x for x in row['observations'] if x['value'].get('event')=='closed']; assert len(closed)==1
                    row['closed_observation'] = closed[0]
                    assert closed[0]['value']['audio_released'] is True and closed[0]['value']['lifetime_active'] is False and closed[0]['value']['stage_detached'] is True
                    await wait_for(lambda:any('CLOUD PREVIEW: returned to full game menu; persistence=false' in x['text'] for x in row['console']))
                    returned = next(x for x in row['console'] if 'CLOUD PREVIEW: returned to full game menu; persistence=false' in x['text'])
                    row['cleanup_precedes_return_log'] = closed[0]['epoch_ms'] <= returned['epoch_ms']; assert row['cleanup_precedes_return_log']
                    assert all(x['value']['host_load_identity']==loaded['host_load_identity'] for x in row['observations'] if x['value'].get('event') in ['loaded','physics'])
                    await page.wait_for_timeout(300); pic = await screen('03-returned-title')
                    row['actual_title_start_pixel_point'] = locate_any(pic,'开始游戏')
                    cmd = await checkpoint('view-real-returned-title',pic,row); assert cmd['action']=='confirm-title' and cmd['passed'] is True
                    row['operator_visual_review'] = cmd['review']
                    row['script_errors'] = [x for x in row['console'] if any(token in x['text'] for token in ['SCRIPT ERROR','Parse Error','Shader compilation','ERROR:'])]
                    assert not row['page_errors'] and not row['script_errors'] and not row['parse_errors']
                    row['status'] = 'PASSED_ACTUAL_PREVIEW_EXIT'
                except Exception as error:
                    row['exception'] = repr(error)
                    try: await screen('failure-preserved')
                    except Exception as screenshot_error: row['failure_screenshot_error'] = repr(screenshot_error)
                    raise
                finally:
                    await context.close(); row['context_closed']=True; save()
        finally:
            await browser.close(); report['browser_closed']=True; save()
    identities = [r['host_load_identity'] for r in report['sessions']]; assert len(set(identities))==1
    report['served_after'] = served_identity(); assert report['served_after']==report['served_before']
    after = guard.snapshot(); report['source_after_raw_sha256'] = compressed('source-after.json.gz',after)
    report['source_stable'] = after==before; assert report['source_stable']
    report['producer_helpers_source_after'] = {name:sha(Path(name)) for name in report['producer_helpers_source_before']}
    report['producer_helpers_stable'] = report['producer_helpers_source_after']==report['producer_helpers_source_before']; assert report['producer_helpers_stable']
    report['elapsed_seconds'] = time.monotonic()-started; assert report['elapsed_seconds']<=300
    report['status']='PASSED_TWO_ACTUAL_WEB_PREVIEW_EXITS'; report.pop('waiting',None); save()

if __name__ == '__main__':
    args = parse_args(); result = {}
    try:
        asyncio.run(asyncio.wait_for(run(args,result),300))
    except BaseException as error:
        result['status']='FAILED'; result['exception']=repr(error)
        if args.output.is_dir(): (args.output/'report.json').write_text(json.dumps(result,ensure_ascii=False,indent=2)+'\n')
        raise
