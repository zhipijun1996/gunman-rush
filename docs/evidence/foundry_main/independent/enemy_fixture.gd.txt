extends SceneTree
var failures:=0
func _initialize()->void:run.call_deferred()
func check(ok:bool,label:String)->void:
	print('ENEMY REVIEW '+label+': '+str(ok));if not ok:failures+=1
func ticks(n:int)->void:
	for i:int in n:await physics_frame
	await process_frame
func run()->void:
	var app:=DemoApp.new();app.meta_persistence_enabled=false;root.add_child(app);app._open_foundry();await ticks(12)
	var lab:FoundryLab=app._foundry;var meta:=app.meta.snapshot();lab.toggle_enemies();await ticks(12)
	for index:int in 24:
		lab.choose(index);await ticks(10)
		if not is_instance_valid(lab.encounter) or lab.encounter.actors.is_empty():continue
		var actor:EnemyActor=lab.encounter.actors[0];var hp:=actor.health.current
		# Explicit combat-only fixture: establish same platform shot position; actual controller/projectile handles damage.
		lab.player.global_position=actor.motor.global_position+Vector2(-65,-2);lab.player.normal_velocity=Vector2.ZERO;lab.player.recoil_velocity=Vector2.ZERO
		lab.controller.router.request_action(&'shoot_release',Vector2.RIGHT);await ticks(20)
		check(actor.health.current<hp,'actual shot damages local enemy '+str(hp)+' to '+str(actor.health.current))
		lab._return();await ticks(5);check(actor.health.current<hp,'return retains injured enemy health')
		var weak:WeakRef=weakref(actor);lab.choose((index+1)%24);await ticks(15);actor=null;check(weak.get_ref()==null,'switch removes old enemy');check(app.meta.snapshot()==meta,'enemy no Meta settlement')
		break
	lab.leave();await ticks(4);app.free();print('ENEMY REVIEW failures='+str(failures));quit(1 if failures else 0)
