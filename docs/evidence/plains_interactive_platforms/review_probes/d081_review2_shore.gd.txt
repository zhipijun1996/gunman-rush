func tick(n:int=1)->void:
 for i:int in n:
  await physics_frame
  c.physics_tick(1.0/60)
  minx=minf(minx,p.position.x);maxx=maxf(maxx,p.position.x)
func run()->void:
 world=Node2D.new();root.add_child(world)
 var module:=PlatformingModule.new();module.definition=preload('res://resources/generation/modules/plains_boss_arena.tres');world.add_child(module)
 p=preload('res://scenes/player/player.tscn').instantiate();p.position=Vector2(266,582);world.add_child(p);c=p.get_node('Controller');c.set_physics_process(false);p.get_node('KeyboardMouseAdapter').set_process_unhandled_input(false);c.enable_shot_energy()
 var boss:=preload('res://scenes/bosses/windkeeper.tscn').instantiate();e=boss.get_node('Encounter');e.definition=preload('res://resources/bosses/windkeeper_storm.tres');boss.position=Vector2(650,535);world.add_child(boss);e.configure_arena(255,1000,p)
 var life:=DemoLifetime.new();var policy:=FrameDamagePolicy.new();policy.lifetime=life;world.add_child(policy);policy.register_target(&'player',c.actor_resources.health,true);policy.bind_damageable(&'boss',boss.get_node('Damageable'),e.health)
 var segment:=SegmentRespawn.new();segment.configure(c,life);e.bind_stage_geometry(module,c,policy,life,segment)
 e.projectile_created.connect(func(s:BossProjectile)->void:
  s.hit.connect(func(body:Node2D,event:StringName,amount:float)->void:
   if body!=p:return
   var r:=DamageRequest.new();r.token=life.token();r.actor_epoch=life.actor_epoch;r.health_epoch=c.actor_resources.health.epoch;r.target_id=&'player';r.source_id=&'boss';r.event_id=event;r.amount=amount;policy.submit(r)))
 await tick(4)
 e.health.apply_damage(ActorResourceRequest.new(&'initial-fixture',e.health.epoch,24,e.health.get_instance_id()));e.activate()
 var prevhp:=e.health.current
 for frame:int in 12000:
  if c.actor_resources.health.terminal:break
  if not pending:c.router.set_move_axis(clampf((266-p.position.x)/5.5,-1,1))
  if not pending and p.is_on_floor() and e.chase_segment==0 and e.direction<0 and e.mode==&'pier_warning' and e.clock<.2:
   pending=true;shotage=0;attempts+=1;c.router.request_action(&'jump')
  if pending:
   shotage+=1
   if shotage==12:c.router.request_action(&'shoot_release',Vector2.DOWN)
   if shotage==30:
    c.router.request_action(&'shoot_release',(boss.position+e.core_offset()-p.position).normalized());print('SHOT attempt=',attempts,' at=',p.position,' core=',boss.position+e.core_offset(),' mode=',e.mode)
   if shotage==45:c.router.request_action(&'jump_release')
   if shotage>60 and p.is_on_floor():pending=false
  await tick()
  if e.health.current<prevhp:hits+=1;print('HIT hp=',e.health.current,' player=',p.position,' boss=',boss.position);prevhp=e.health.current
  if hits>=2:break
 print('RESULT hits=',hits,' attempts=',attempts,' playerhp=',c.actor_resources.health.current,' xspan=',minx,'..',maxx,' bosshp=',e.health.current,' phase=',e.stage_phase,' mode=',e.mode)
 world.free();quit()
