extends SceneTree
var checks:=0
var failed:=0
func _initialize()->void:
 var g:=PlainsRegionGenerator.new()
 for i:int in 30:
  var r:Dictionary=g.generate('review4-port/'+str(i),PlayerTuning.load_default(),false,'','intro' if i<15 else 'advanced')
  check(r.ok,'actual port metadata map generates')
  if not r.ok:continue
  var m:Dictionary=r.manifest
  check(g.validate(JSON.parse_string(JSON.stringify(m)),PlayerTuning.load_default()),'port metadata survives exact JSON replay validation')
  for edge:Dictionary in m.connections:
   if edge.kind!='horizontal':continue
   if edge.recipe not in ['controlled_ferry','remote_bell']:continue
   var actions:Dictionary=g._connection_actions(m,edge)
   check(actions.shoot_mechanism if edge.recipe=='controlled_ferry' else actions.get('optional_remote_shot',false) and not actions.shoot_mechanism,'actual ferry required/upper detour optional shot marked')
   for a:Dictionary in m.areas:
    for port:Dictionary in a.ports:
     if port.connection==edge.id:check(port.actions==actions,'actual area port consumes correct action metadata')
 print('REVIEW4 PORTS assertions=',checks,' failures=',failed)
 quit(1 if failed else 0)
func check(ok:bool,msg:String)->void:
 checks+=1
 if not ok:failed+=1;printerr('FAIL '+msg)
